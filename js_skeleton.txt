7345:     function submitLearnGoal() {
7346:       const what = document.getElementById('learnWhat').value.trim();
7347:       if (!what) { alert('说说你想学什么吧～'); return; }
7348: 
7349:       const duration = document.getElementById('learnDuration').value.trim();
7350:       const timePerDay = document.getElementById('learnTimePerDay').value.trim();
7351:       const treeId = document.getElementById('learnTreeId').value;
7352: 
7353:       const fullText = what + ' ' + duration + ' ' + timePerDay;
7354:       const assumptions = [];
7355: 
7356:       // ---- 提取时长（缺省做保守假设并标注） ----
7357:       let days = 0;
7358:       const dayMatch = fullText.match(/(\d+)\s*[天日]/);
7359:       const monthMatch = fullText.match(/(\d+)\s*个?月/);
7360:       const weekMatch = fullText.match(/(\d+)\s*个?星期|(\d+)\s*周/);
7361:       if (dayMatch) days = parseInt(dayMatch[1]);
7362:       else if (monthMatch) days = parseInt(monthMatch[1]) * 30;
7363:       else if (weekMatch) days = parseInt(weekMatch[1] || weekMatch[2]) * 7;
7364:       if (!days && duration) {
7365:         const numMatch = duration.match(/(\d+)/);
7366:         if (numMatch) days = parseInt(numMatch[1]);
7367:       }
7368:       if (!days) {
7369:         days = 30;
7370:         assumptions.push('时长没写，先按 30 天排——想改的话，在目标里写上"3个月 / 两周"重新生成就好');
7371:       }
7372: 
7373:       // ---- 提取每天时间 ----
7374:       let minutesPerDay = 0;
7375:       const hourMatch = fullText.match(/(\d+(?:\.\d+)?)\s*个?小时/);
7376:       const minMatch = fullText.match(/(\d+)\s*分钟/);
7377:       if (hourMatch) minutesPerDay = Math.round(parseFloat(hourMatch[1]) * 60);
7378:       else if (minMatch) minutesPerDay = parseInt(minMatch[1]);
7379:       if (!minutesPerDay) {
7380:         minutesPerDay = 60;
7381:         assumptions.push('每天投入时间先按 1 小时排');
7382:       }
7383: 
7384:       // ---- 提取基础 ----
7385:       let isZero = /零基础|完全不会|没接触过|没用过|小白|从零/.test(fullText);
7386:       const hasExperience = !isZero && /有一点|有一些|用过|学过一点|会一点|有点基础|学过/.test(fullText);
7387:       const wantsAdvanced = /精通|专业级|高级|深入|系统/.test(fullText);
7388:       if (!isZero && !hasExperience) {
7389:         // 与"保守假设"标注保持一致：默认按零基础起步
7390:         isZero = true;
7391:         assumptions.push('基础先按零基础起步排（有基础的话进度会更快）');
7392:       }
7393: 
7394:       // ---- 匹配领域，直接生成 ----
7395:       const domain = matchDomain(fullText);
7396:       const skillName = domain ? domain.name : extractSkillName(what);
7397:       const unitDefs = buildUnitDefs(domain, { isZero, hasExperience, wantsAdvanced, what });
7398:       const verifyTask = domain ? domain.verifyTask : genericVerifyTask(what);
7399:       const plan = aiGeneratePlan(unitDefs, days, minutesPerDay, verifyTask);
7400: 
7401:       // ---- 保存技能 ----
7402:       const skills = getSkills();
7403:       let finalName = skillName;
7404:       if (skills.find(s => s.name === finalName)) finalName = skillName + '（' + today() + '）';
7405: 
7406:       skills.push({
7407:         name: finalName,
7408:         status: 'learning',
7409:         sourceExpIds: [],
7410:         confirmedAt: null,
7411:         goalText: what,
7412:         goalDuration: duration || (days + '天'),
7413:         goalTreeId: treeId || null,
7414:         assumptions: assumptions,
7415:         breakdown: unitDefs.map((u, i) => ({
7416:           id: 'u' + i,
7417:           name: u.name,
7418:           phase: u.phase,
7419:           status: 'want',
7420:           sourceExpId: null
7421:         })),
7422:         plan: plan,
7423:         verifyTask: verifyTask,
7424:         treeIds: treeId ? [treeId] : []
7425:       });
7426:       saveSkills(skills);
7427: 
7428:       closeLearnGoalModal();
7429:       renderTreeTabs();

7749:       const t = text.toLowerCase();
7750:       for (const d of DOMAIN_LIB) {
7751:         if (d.patterns.some(p => p.test(t))) return d;
7752:       }
7753:       return null;

7756:     // 根据领域 + 用户情况生成能力单元定义（含任务池）
7757:     function buildUnitDefs(domain, opts) {
7758:       let defs;
7759:       if (domain) {
7760:         defs = domain.units.filter(u =>
7761:           (!u.adv || opts.wantsAdvanced) &&
7762:           (!u.beginnerOnly || opts.isZero)
7763:         ).map(u => ({ name: u.name, tasks: u.tasks }));
7764:       } else {
7765:         defs = genericUnits(opts.what);
7766:       }
7767:       // 分阶段：前 40% 打基础，中间核心练习，最后整合实战
7768:       const n = defs.length;
7769:       defs.forEach((u, i) => {
7770:         u.phase = i < n * 0.4 ? 1 : (i < n * 0.85 ? 2 : 3);
7771:       });
7772:       return defs;
7773:     }

7817:       if (/英语口语|口语/.test(text)) return '英语口语';

7819:       if (/健身|跑步|减脂|增肌/.test(text)) return '健身';

7827:       const colonMatch = what.match(/[：:]([^，。,!?！？\s]{1,12})/);
7828:       if (colonMatch) return colonMatch[1].trim();

7830:       // 尝试提取"学会XX / 学个XX / 学一下XX"中的XX
7831:       const match = what.match(/学[会习]?(?:个|一下)?([^，。,!?！？\s]{1,12}?)(?:，|。|,|。|\s|$)/);

7835:       const stripped = what.replace(/^我[想要]+/, '').replace(/用?\d+\s*[天日周月]*/, '').replace(/^[里内的]+/, '');

7839:     /* ============================================================
7840:        16) 学习计划生成：先分阶段，再排天/周任务
7841:        ≤45 天 → 按天（Day N）；>45 天 → 按周（Week N，每周 5 个任务）
7842:        任务全部来自领域知识库，保证"具体、能做、有结果"
7843:        ============================================================ */
7844:     function aiGeneratePlan(unitDefs, totalDays, minutesPerDay, verifyTask) {
7845:       const plan = [];
7846:       const defTime = minutesPerDay + '分钟';
7847:       const dailyMode = totalDays <= 45;
7848:       const totalWeight = unitDefs.reduce((s, u) => s + u.tasks.length, 0) || 1;
7849:       const verifyTime = Math.max(minutesPerDay * 2, 120) + '分钟';
7850: 
7851:       if (dailyMode) {
7852:         const learningDays = Math.max(unitDefs.length, totalDays - 1);
7853:         // 按任务量分配天数，但每个单元最多"独立任务数 + 2"天，
7854:         // 避免同一个练习机械重复太多次；多出/不足的部分按比例再平衡
7855:         const alloc = [];
7856:         let allocated = 0;
7857:         unitDefs.forEach((u, idx) => {
7858:           const unique = u.tasks.filter(t => !t.repeat).length || u.tasks.length;
7859:           const raw = Math.max(1, Math.round(learningDays * u.tasks.length / totalWeight));
7860:           alloc[idx] = Math.min(raw, unique + 2);
7861:           allocated += alloc[idx];
7862:         });
7863:         if (allocated !== learningDays) {
7864:           if (allocated > learningDays) {
7865:             // 超了：按比例缩
7866:             const scale = learningDays / allocated;
7867:             let sum = 0;
7868:             unitDefs.forEach((u, idx) => { alloc[idx] = Math.max(1, Math.round(alloc[idx] * scale)); sum += alloc[idx]; });
7869:             allocated = sum;
7870:           }
7871:           // 补齐或继续削减
7872:           let rest = learningDays - allocated, i = 0, guard = 0;
7873:           while (rest !== 0 && guard < 2000) {
7874:             const k = i % unitDefs.length;
7875:             if (rest > 0) { alloc[k]++; rest--; }
7876:             else if (alloc[k] > 1) { alloc[k]--; rest++; }
7877:             i++; guard++;
7878:           }
7879:         }
7880:         let day = 1;
7881:         unitDefs.forEach((u, idx) => {
7882:           const unitId = 'u' + idx;
7883:           const counts = {};
7884:           expandTasks(u.tasks, alloc[idx], defTime, 0, counts).forEach(t => {
7885:             plan.push({ unitId, day, title: 'Day ' + day + '｜' + u.name, do: t.do, time: t.time, standard: t.standard, done: false });
7886:             day++;
7887:           });
7888:         });
7889:         plan.push({
7890:           unitId: 'verify', day: day, title: 'Day ' + day + '｜🏆 验证任务',
7891:           do: verifyTask, time: verifyTime,
7892:           standard: '完成并记录为一段真实经历（记录后技能正式点亮）', done: false
7893:         });
7894:       } else {
7895:         // 周模式：先分阶段，每周 5 个任务
7896:         const weeks = Math.max(unitDefs.length + 1, Math.round(totalDays / 7));
7897:         let week = 1;
7898:         unitDefs.forEach((u, idx) => {
7899:           const unitId = 'u' + idx;
7900:           let unitWeeks = Math.max(1, Math.round((weeks - 1) * u.tasks.length / totalWeight));
7901:           if (idx === unitDefs.length - 1) unitWeeks = Math.max(1, (weeks - 1) - (week - 1));
7902:           let offset = 0;
7903:           const counts = {}; // 跨周累计使用次数
7904:           for (let w = 0; w < unitWeeks; w++) {
7905:             expandTasks(u.tasks, 5, defTime, offset, counts).forEach(t => {
7906:               plan.push({ unitId, day: week, title: 'Week ' + week + '｜' + u.name, do: t.do, time: t.time, standard: t.standard, done: false });
7907:             });
7908:             offset += 5;
7909:             week++;
7910:           }
7911:         });
7912:         plan.push({
7913:           unitId: 'verify', day: week, title: 'Week ' + week + '｜🏆 验证任务',
7914:           do: verifyTask, time: verifyTime,
7915:           standard: '完成并记录为一段真实经历（记录后技能正式点亮）', done: false
7916:         });
7917:       }
7918: 
7919:       return plan;
7920:     }
7921: 
7922:     // 从任务池展开 N 个任务：先消耗一遍全部任务，
7923:     // 耗尽后循环整个池子（可重复练习的任务权重 ×2），并标注"第N次"
7924:     function expandTasks(pool, count, defTime, offset, counts) {
7925:       const out = [];
7926:       if (!pool.length) return out;
7927:       const firstLen = pool.length;
7928:       const cyc = [];
7929:       pool.forEach((t, i) => { cyc.push(i); if (t.repeat) cyc.push(i); });
7930:       const c = counts || {};
7931:       for (let j = 0; j < count; j++) {
7932:         const slot = (offset || 0) + j;
7933:         const idx = slot < firstLen ? slot : cyc[(slot - firstLen) % cyc.length];
7934:         c[idx] = (c[idx] || 0) + 1;
7935:         const t = pool[idx];
7936:         const suffix = c[idx] > 1 ? '（第' + c[idx] + '次）' : '';
7937:         out.push({ do: t.do + suffix, time: t.time || defTime, standard: t.standard });
7938:       }
7939:       return out;
7940:     }
